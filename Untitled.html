<script>

    const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzfe4GMTcSBLMMAtnut9RDtMd7IsGmKOZREVd47vRkZzuiwk3pOLlf3KhKcevsW0826qg/exec";


    const config = {

        employees: [

            { id: "EMP-001", name: “HERMAWAN” },

            { id: "EMP-002", name: “UHRAN” },

            { id: "EMP-003", name: “HADIANSAHI" },

            { id: "EMP-004", name: “RISDA “ },

            { id: "EMP-005", name: “ZUL”  },

            { id: "EMP-006", name: “DINO” },

            { id: "EMP-007", name: "NURADIAN PUTRI" },

            { id: "EMP-007", name: “HERSONIMUS” },

            { id: "EMP-007", name: “RICARD” },

            { id: "EMP-008", name: "SARIPAH" }

        ]

    };


    let videoStream = null;


    document.addEventListener("DOMContentLoaded", () => {

        initApp();

    });


    function initApp() {

        renderEmployeeOptions();

        updateClock();

        setInterval(updateClock, 1000);

        getLocation();

        initCamera();

    }


    function renderEmployeeOptions() {

        const select = document.getElementById("employeeSelect");

        config.employees.forEach(emp => {

            const opt = document.createElement("option");

            opt.value = emp.id;

            opt.textContent = emp.name;

            select.appendChild(opt);

        });

    }


    function autoFillEmployeeId() {

        const select = document.getElementById("employeeSelect");

        document.getElementById("employeeId").value = select.value;

    }


    function updateClock() {

        const now = new Date();

        document.getElementById("liveClock").textContent = now.toLocaleTimeString('id-ID');

        document.getElementById("liveDate").textContent = now.toLocaleDateString('id-ID', {

            weekday: 'long',

            year: 'numeric',

            month: 'long',

            day: 'numeric'

        });

    }


    function getLocation() {

        const status = document.getElementById("locationStatus");

        const coordsInput = document.getElementById("locationCoords");


        if (navigator.geolocation) {

            navigator.geolocation.getCurrentPosition(

                (position) => {

                    const lat = position.coords.latitude;

                    const lng = position.coords.longitude;

                    coordsInput.value = `${lat}, ${lng}`;

                    status.textContent = "Lokasi terdeteksi dengan akurat.";

                    status.className = "text-success";

                },

                (error) => {

                    status.textContent = "Gagal mengambil lokasi. Pastikan izin GPS aktif.";

                    status.className = "text-danger";

                },

                { enableHighAccuracy: true }

            );

        } else {

            status.textContent = "Browser Anda tidak mendukung Geolocation.";

            status.className = "text-danger";

        }

    }


    async function initCamera() {

        const video = document.getElementById("webcam");

        try {

            videoStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: false });

            video.srcObject = videoStream;

        } catch (err) {

            alert("Tidak dapat mengakses kamera. Pastikan memberikan izin pada browser.");

        }

    }


    function takeSnapshot() {

        const video = document.getElementById("webcam");

        const canvas = document.getElementById("photoCanvas");

        const photoData = document.getElementById("photoData");

        const context = canvas.getContext("2d");


        canvas.width = video.videoWidth || 320;

        canvas.height = video.videoHeight || 240;


        context.translate(canvas.width, 0);

        context.scale(-1, 1);

        context.drawImage(video, 0, 0, canvas.width, canvas.height);


        const dataURL = canvas.toDataURL("image/jpeg");

        photoData.value = dataURL;


        video.style.display = "none";

        canvas.style.display = "block";

        document.getElementById("btnCapture").style.display = "none";

        document.getElementById("btnRetake").style.display = "inline-block";

    }


    function resetCamera() {

        document.getElementById("webcam").style.display = "block";

        document.getElementById("photoCanvas").style.display = "none";

        document.getElementById("btnCapture").style.display = "inline-block";

        document.getElementById("btnRetake").style.display = "none";

        document.getElementById("photoData").value = "";

    }


    document.getElementById("attendanceForm").addEventListener("submit", function (e) {

        e.preventDefault();


        const photo = document.getElementById("photoData").value;

        if (!photo) {

            alert("Harap ambil foto selfie terlebih dahulu!");

            return;

        }


        const btnSubmit = document.getElementById("btnSubmit");

        btnSubmit.disabled = true;

        btnSubmit.textContent = "Mengirim...";


        const payload = {

            employeeId: document.getElementById("employeeId").value,

            employeeName: document.getElementById("employeeSelect").options[document.getElementById("employeeSelect").selectedIndex].text,

            type: document.querySelector('input[name="attendanceType"]:checked').value,

            location: document.getElementById("locationCoords").value,

            photo: photo

        };


        fetch(GOOGLE_SCRIPT_URL, {

            method: "POST",

            mode: "no-cors",

            headers: { "Content-Type": "application/json" },

            body: JSON.stringify(payload)

        })

        .then(() => {

            alert("Presensi berhasil terkirim!");

            location.reload();

        })

        .catch((error) => {

            alert("Terjadi kesalahan saat mengunggah data.");

            btnSubmit.disabled = false;

            btnSubmit.textContent = "Kirim Presensi";

        });

    });
